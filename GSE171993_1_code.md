## calculate pairwise QC correlations
- can run by libID as well
```r
library(Seurat)
library(ggplot2)
library(patchwork)

work_dir <- "/scratch/delphine/neutrophils"
seu <- readRDS(file.path(work_dir, "GSE171993_postQC_8aug.rds"))

df <- seu@meta.data
n_cells <- format(nrow(df), big.mark = ",")

pairs <- list(
c("nCount_RNA", "nFeature_RNA"),
c("nCount_RNA", "percent.mt"),
c("nCount_RNA", "percent.ribo"),
c("nFeature_RNA", "percent.mt"),
c("nFeature_RNA", "percent.ribo"),
c("percent.ribo", "percent.mt")
)

plot_list <- list()

for (i in seq_along(pairs)) {
var_x <- pairs[[i]][1]
var_y <- pairs[[i]][2]

ct <- cor.test(df[[var_x]], df[[var_y]], method = "pearson")
r_str <- sprintf("%.2f", ct$estimate)
p_str <- if (ct$p.value < 0.01) "< 0.01" else sprintf("%.2f", ct$p.value)

title_label <- paste0("r = ", r_str, ", p ", p_str)

p <- ggplot(df, aes(x = .data[[var_x]], y = .data[[var_y]])) +
geom_point(color = "#34495e", alpha = 0.2, size = 0.4) +
labs(title = title_label, x = var_x, y = var_y) +
theme_classic(base_size = 13) +
theme(
plot.title = element_text(hjust = 0.5, face = "bold", size = 15),
axis.title = element_text(size = 13, face = "bold"),
axis.text = element_text(size = 11, color = "black"),
axis.text.x = element_text(angle = 45, hjust = 1),
plot.margin = margin(t = 10, r = 10, b = 25, l = 10)
)

plot_list[[i]] <- p
}

combined_plot <- wrap_plots(plot_list, ncol = 2) +
plot_annotation(
title = "Comprehensive Pairwise QC Metrics",
subtitle = paste0("All Cells, n = ", n_cells),
theme = theme(
plot.title = element_text(size = 18, face = "bold", hjust = 0.5),
plot.subtitle = element_text(size = 14, face = "plain", hjust = 0.5)
)
)

timestamp <- format(Sys.time(), "%Y%m%d_%H%M%S")
png_path <- file.path(work_dir, paste0("GSE171993_comprehensive_qc_scatters_2col_", timestamp, ".png"))
ggsave(png_path, plot = combined_plot, width = 9, height = 13, dpi = 300)

cat("plot successfully saved to:", png_path, "\n")

library(SoupX)
library(Seurat)
library(ggplot2)

work_dir <- "/scratch/delphine/neutrophils"
seu <- readRDS(file.path(work_dir, "GSE171993_postQC_8aug.rds"))

toc <- LayerData(seu, layer = "counts")

sc <- SoupChannel(toc, toc, calcSoupProfile = FALSE)

soupProf <- data.frame(
row.names = rownames(toc),
est = rowSums(toc) / sum(toc),
counts = rowSums(toc)
)
sc <- setSoupProfile(sc, soupProf)

sc <- setClusters(sc, setNames(as.character(seu$celltype), colnames(seu)))

timestamp <- format(Sys.time(), "%Y%m%d_%H%M%S")
png_path <- file.path(work_dir, paste0("GSE171993_soupx_rho_plot_", timestamp, ".png"))

png(filename = png_path, width = 8, height = 6, units = "in", res = 300)
sc <- autoEstCont(sc, doPlot = TRUE)
dev.off()

rho_global <- sc$fit$rhoEst
rho_percent <- sprintf("%.2f%%", rho_global * 100)
n_cells <- ncol(seu)
n_genes <- nrow(seu)
n_clusters <- length(unique(seu$celltype))

metrics_df <- data.frame(
Metric = c(
"Dataset",
"Total_Cells",
"Total_Genes",
"Total_CellTypes",
"Global_Rho_Raw",
"Global_Rho_Percent"
),
Value = c(
"GSE171993",
as.character(n_cells),
as.character(n_genes),
as.character(n_clusters),
sprintf("%.4f", rho_global),
rho_percent
)
)

tsv_path <- file.path(work_dir, paste0("GSE171993_soupx_rho_metrics_", timestamp, ".tsv"))
write.table(metrics_df, file = tsv_path, sep = "\t", quote = FALSE, row.names = FALSE)

markers_df <- sc$fit$markersUsed

if (!is.null(markers_df) && nrow(markers_df) > 0) {
markers_df <- markers_df[order(markers_df$cluster, -markers_df$tfidf), ]
}

markers_tsv_path <- file.path(work_dir, paste0("GSE171993_soupx_top_markers_", timestamp, ".tsv"))
write.table(markers_df, file = markers_tsv_path, sep = "\t", quote = FALSE, row.names = FALSE)

cat("\nSoupX Estimation Summary:\n")
print(metrics_df, row.names = FALSE)

cat("\nTop Estimation Markers and Specificity (TF-IDF) Per Cell Type:\n")
if (!is.null(markers_df) && nrow(markers_df) > 0) {
split_markers <- split(markers_df, markers_df$cluster)
for (ct in names(split_markers)) {
top_df <- head(split_markers[[ct]], 3)
marker_info <- paste0(top_df$gene, " (tfidf=", round(top_df$tfidf, 2), ")")
cat(sprintf("  %s: %s\n", ct, paste(marker_info, collapse = ", ")))
}
} else {
cat("  no specific markers table returned in sc$fit$markersUsed\n")
}

cat("\nplot successfully saved to:", png_path, "\n")
cat("summary metrics successfully saved to:", tsv_path, "\n")
cat("markers table successfully saved to:", markers_tsv_path, "\n")
```

# scDblFinder remove doublets
- set serial multithread to prevent stalling 
```r 
library(Seurat)
library(scDblFinder)
library(SingleCellExperiment)
library(BiocParallel)
library(ggplot2)
library(tidyr)

work_dir <- "/scratch/delphine/neutrophils"
seu <- readRDS(file.path(work_dir, "GSE171993_postQC_8aug.rds"))

seu$lib_ID <- factor(seu$lib_ID, levels = as.character(1:10))

sce <- as.SingleCellExperiment(seu)

bp <- SerialParam(RNGseed = 1234)

sce <- scDblFinder(sce, samples = "lib_ID", BPPARAM = bp, verbose = TRUE)

seu$scDblFinder.score <- sce$scDblFinder.score
seu$scDblFinder.class <- sce$scDblFinder.class

timestamp <- format(Sys.time(), "%Y%m%d_%H%M%S")

plot_df <- data.frame(
lib_ID = seu$lib_ID,
lib = seu$lib,
score = seu$scDblFinder.score,
class = seu$scDblFinder.class
)

p <- ggplot(plot_df, aes(x = score, fill = class)) +
geom_histogram(bins = 50, color = "black", linewidth = 0.2, alpha = 0.7) +
facet_wrap(~ lib_ID + lib, scales = "free_y", ncol = 5) +
scale_fill_manual(values = c("singlet" = "#377EB8", "doublet" = "#E41A1C")) +
theme_minimal(base_size = 11) +
labs(
title = "scDblFinder Score Distribution Per Sample",
x = "Doublet Score",
y = "Cell Count",
fill = "Classification"
) +
theme(
strip.background = element_rect(fill = "grey90", color = NA),
legend.position = "bottom"
)

png_path <- file.path(work_dir, paste0("GSE171993_scDblFinder_diagnostics_", timestamp, ".png"))
ggsave(png_path, plot = p, width = 14, height = 7, dpi = 300)

summary_tbl <- as.data.frame(table(seu$lib_ID, seu$lib, seu$scDblFinder.class))
colnames(summary_tbl) <- c("lib_ID", "lib", "Class", "Count")

summary_tbl <- summary_tbl[summary_tbl$Count > 0, ]
summary_wide <- pivot_wider(summary_tbl, names_from = Class, values_from = Count, values_fill = 0)

summary_wide$lib_ID <- as.numeric(as.character(summary_wide$lib_ID))
summary_wide <- summary_wide[order(summary_wide$lib_ID), ]

summary_wide$Total_Cells <- summary_wide$singlet + summary_wide$doublet
summary_wide$Doublet_Rate <- sprintf("%.2f%%", (summary_wide$doublet / summary_wide$Total_Cells) * 100)

tsv_path <- file.path(work_dir, paste0("GSE171993_scDblFinder_attrition_", timestamp, ".tsv"))
write.table(summary_wide, file = tsv_path, sep = "\t", quote = FALSE, row.names = FALSE)

saveRDS(seu, file.path(work_dir, "GSE171993_scDblFinder_annotated.rds"))

cat("\nscDblFinder Sample Attrition Assessment:\n")
print(as.data.frame(summary_wide), row.names = FALSE)
cat("\ndiagnostic histogram saved to:", png_path, "\n")
cat("attrition table saved to:", tsv_path, "\n")
```

## quantile attrition 98 vs 99

```r
setwd("/scratch/delphine/neutrophils")

library(dplyr)

options(width = 500)

rds_path <- "GSE171993_singlets_9aug.rds"
seu_singlets <- readRDS(rds_path)

total_cells <- ncol(seu_singlets)

cat("File:", rds_path, "\n")
cat("Total cells:", total_cells, "\n\n")

# calculate percentiles
q98 <- quantile(seu_singlets$nCount_RNA, 0.98, na.rm = TRUE)
q99 <- quantile(seu_singlets$nCount_RNA, 0.99, na.rm = TRUE)

# calculate passing cell counts and attrition
pass_98 <- sum(seu_singlets$nCount_RNA <= q98)
attr_cnt_98 <- total_cells - pass_98
attr_pct_98 <- (attr_cnt_98 / total_cells) * 100

pass_99 <- sum(seu_singlets$nCount_RNA <= q99)
attr_cnt_99 <- total_cells - pass_99
attr_pct_99 <- (attr_cnt_99 / total_cells) * 100

# format summary tables
summary_q98 <- data.frame(
Scenario = "98th Percentile",
nCount_Cutoff = round(q98, 2),
Final_Cell_Count = pass_98,
Attrition_Cells = attr_cnt_98,
Attrition_Percent = paste0(round(attr_pct_98, 2), "%")
)

summary_q99 <- data.frame(
Scenario = "99th Percentile",
nCount_Cutoff = round(q99, 2),
Final_Cell_Count = pass_99,
Attrition_Cells = attr_cnt_99,
Attrition_Percent = paste0(round(attr_pct_99, 2), "%")
)

attrition_summary <- rbind(summary_q98, summary_q99)
print(attrition_summary)

# export summary objects
saveRDS(summary_q98, "GSE171993_attrition_summary_q98.rds")
saveRDS(summary_q99, "GSE171993_attrition_summary_q99.rds")
```
